SELECT
	YEAR(H.OrderDate) AS Ano,
	MONTH(H.OrderDate) AS Mes,
	SUM(D.LineTotal) AS Receita
FROM 
		AdventureWorks2016.sales.SalesOrderDetail AS D
		INNER JOIN AdventureWorks2016.sales.SalesOrderHeader AS H
		ON D.SalesOrderID = H.SalesOrderID
WHERE
	H.OrderDate >= '20130101'
	AND H.OrderDate < '20130701'
GROUP BY
	YEAR(H.OrderDate),
	MONTH(H.OrderDate)
ORDER BY
Ano, Mes
